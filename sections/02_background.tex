\begin{frame}{Global-Mask Compression}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \textbf{Compressor state}
      \begin{itemize}
        \item Token sequence $x_{1:n}$
        \item Global allowed-token set $V'$
        \item Row seeds for batched streams
        \item Context limit $R$
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \textbf{Per-step operation}
      \begin{align*}
        z_t &= M(c_t) \\
        p_j &= \frac{\exp z_{t,j}}{\sum_{k \in V'} \exp z_{t,k}} \\
        \ell_t &\approx -\log_2 p_{x_{t+1}}
      \end{align*}
    \end{column}
  \end{columns}

  \vspace{0.7em}
  \centering
  \small Lossless replay requires the same model, tokenizer, vocabulary mask, context policy, and coder parameters.
\end{frame}
