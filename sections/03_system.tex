\begin{frame}{Three Optimizations Target Three Costs}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.31\textwidth}
      \begin{block}{O1: vLLM windows}
        Teacher-force multiple known target tokens in one engine call.

        \vspace{0.3em}
        \small Targets request scheduling overhead.
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.31\textwidth}
      \begin{block}{O2: multistream AC}
        Maintain one range coder per batch row and compile the hot update loop \parencite{Lam2015NumbaAL}.

        \vspace{0.3em}
        \small Targets Python and coder overhead.
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.31\textwidth}
      \begin{block}{O3: target interval}
        Return only the interval needed for the true next token.

        \vspace{0.3em}
        \small Targets score conversion and transfer.
      \end{block}
    \end{column}
  \end{columns}
\end{frame}
