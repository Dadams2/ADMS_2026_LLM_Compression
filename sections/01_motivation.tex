\section{Motivation}

\begin{frame}{Text Is a First-Class Analytics Datatype}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.48\textwidth}
      \begin{block}{Amazon Redshift fleet}
        \centering
        {\Huge 52.1\%}\\[0.4em]
        \small Stored columns observed as \texttt{varchar}\\[-0.1em]
        \scriptsize\parencite{vanrenenWhyTPCNot2024}
      \end{block}
    \end{column}
    \hfill
    \begin{column}{0.48\textwidth}
      \begin{block}{Snowflake workloads}
        \centering
        {\Huge 34\%}\\[0.4em]
        \small Aggregation inputs reported as text columns\\[-0.1em]
        \scriptsize\parencite{szlangWorkloadInsightsSnowflake2025}
      \end{block}
    \end{column}
  \end{columns}

  \vspace{1.0em}
  \begin{itemize}
    \item Storage is expensive 
    \item Data is stored for backup or compliance
    \item Most data is \textbf{cold} the moment it is created
  \end{itemize}
\end{frame}

\begin{frame}{Storage Codec Question: Current Practice}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.50\textwidth}
      \textbf{Block-based compression}
      \begin{itemize}
        \item Compress independent byte blocks
        \item Cheap CPU path at ingest and scan time
        \item Mature formats: \codec{zstd}, FSST, dictionary codecs \parencite{zstd,fsst}
        \item Good engineering fit for DBMS storage pages
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.44\textwidth}
      \begin{block}{Why it works operationally}
        \centering
        \Large fast, cheap, predictable

        \vspace{0.6em}
        \normalsize
        The codec adds little latency and keeps decompression local to the block being read.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.8em}
  \centering
  \small The tradeoff: classical codecs leave language-level redundancy on the table.
\end{frame}

\begin{frame}{Storage Codec Question: Language Models as Compressors}
  \begin{columns}[T, onlytextwidth]
    \begin{column}{0.52\textwidth}
      \textbf{Prediction becomes compression}
      \begin{itemize}
        \item Better next-token probabilities mean fewer encoded bits
        \item Chinchilla 70B beats domain-specific compressors on several modalities \parencite{deletangLanguageModelingCompression2024}
        \item LLMZip and later work show strong text ratios \parencite{llmzip,lossless_data_compression_llm}
      \end{itemize}
    \end{column}
    \hfill
    \begin{column}{0.42\textwidth}
      \begin{block}{Systems catch}
        \centering
        \Large slow unless engineered

        \vspace{0.6em}
        \normalsize
        Encode and decode replay the same model calls and entropy-coder updates.
      \end{block}
    \end{column}
  \end{columns}

  \vspace{0.9em}
  \begin{block}{Central systems question}
    Under which combinations of inference engine, hardware platform, and entropy-coding design can LLM compression become economically viable for analytical text storage?
  \end{block}
\end{frame}

\begin{frame}[standout]
  \Large Compression ratio is not enough.

  \vspace{1.0em}
  \normalsize
  A database codec has to balance storage footprint, ingestion throughput, and decode cost \parencite{Kipf2026WaitingTD}.
\end{frame}
