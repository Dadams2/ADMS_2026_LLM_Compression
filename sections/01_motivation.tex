\begin{frame}{Compression performance scales with model size}

  \small
  \textbf{Raw compression rate (\%)} \quad compressed size / original size; lower is better

  \setlength{\tabcolsep}{10pt}
  \begin{tabular}{lccc}
    \toprule

    \midrule
    gzip            & 48.1 & 68.6 & 38.5 \\

    FLAC            & 88.9 & 60.9 & 30.3 \\
    \midrule
    Chinchilla 1B   & 11.3 & 62.2 & 24.9 \\
    Chinchilla 7B   & 10.2 & 54.7 & 23.6 \\
    \rowcolor{mDarkTeal!12}
    \textbf{Chinchilla 70B} & \textbf{8.3} & \textbf{48.0} & \textbf{21.0} \\
    \bottomrule
  \end{tabular}

  \vspace{0.2em}
  \footnotesize \textbf{\textcolor{mDarkTeal}{Takeaway:}} A text-trained LLM can compress image and audio bytes.

  \scriptsize 1~GB datasets; 2,048-byte chunks; model weights excluded. Adjusted Chinchilla rates are 211--14,048\% \parencite{deletangLanguageModelingCompression2024}.
